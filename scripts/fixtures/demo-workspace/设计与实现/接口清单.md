              type: 'string',
              desc: '按状态筛选（pending/paid/shipped/done/closed），缺省返回全部',
            }
          - { name: 'page', in: 'query', type: 'number', desc: '页码，缺省 1' }
          - { name: 'pageSize', in: 'query', type: 'number', desc: '每页条数，上限 100，缺省 20' }
        response:
          - { name: 'items', type: 'OrderBrief[]', desc: '订单摘要列表（不含订单项明细）' }
          - { name: 'total', type: 'number', desc: '符合筛选的总条数' }

        path: /api/orders
        summary: 创建订单：校验并锁定库存，金额以服务端快照为准
        auth: 登录
        params:
          - {
              name: 'items',
              in: 'body',
              type: 'OrderItemInput[]',
              required: true,
              desc: '订单项（商品 ID + 数量）；同一商品重复出现会被合并',
            }
          - { name: 'remark', in: 'body', type: 'string', desc: '买家备注，上限 200 字' }
        response:
          - { name: 'orderId', type: 'bigint', desc: '新订单 ID' }
          - { name: 'status', type: 'string', desc: '初始状态恒为 pending' }
          - { name: 'amount', type: 'decimal(10,2)', desc: '服务端按当下价计算的总额' }
        invariants:
          - '金额只由服务端计算：客户端提交的价格字段不参与结算'
          - statement: '库存不足时整单拒绝（不部分成交、不超卖）'
            kind: must-not
        note: '创建成功后同事务落 order_items；支付超时由定时任务关单并回滚库存。'
      - method: POST
        path: /api/orders/:id/cancel
        summary: 取消订单：仅待支付状态可取消，库存同事务回滚
        auth: 登录
        params:
          - { name: 'id', in: 'path', type: 'bigint', required: true, desc: '订单 ID' }

          - { name: 'status', type: 'string', desc: '取消后恒为 closed' }
